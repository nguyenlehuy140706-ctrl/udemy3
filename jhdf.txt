<script>
function searchCourses() {
    const keyword = document.getElementById("searchInput")
        .value.toLowerCase().trim();

    let count = 0;

    document.querySelectorAll(".course").forEach(course => {
        const matches = course.innerText.toLowerCase().includes(keyword);
        const category = course.dataset.category;
        const activeCategory = document.querySelector(".category.active");
        const selected = activeCategory.dataset.filter || "all";

        const visible = matches &&
            (selected === "all" || selected === category);

        course.style.display = visible ? "" : "none";

        if (visible) count++;
    });

    document.getElementById("noResults").style.display =
        count === 0 ? "block" : "none";
}

function filterCourses(category, button) {
    document.querySelectorAll(".category").forEach(item => {
        item.classList.remove("active");
        item.dataset.filter = "";
    });

    button.classList.add("active");
    button.dataset.filter = category;

    document.querySelectorAll(".course").forEach(course => {
        const matches = category === "all" ||
            course.dataset.category === category;

        course.style.display = matches ? "" : "none";
    });

    document.getElementById("noResults").style.display =
        document.querySelectorAll(".course").length > 0 &&
        [...document.querySelectorAll(".course")]
            .every(course => course.style.display === "none")
            ? "block" : "none";

    searchCourses();
}

function enroll(courseName) {
    alert("Bạn đã chọn khóa học: " + courseName +
        "\nVui lòng đăng nhập để tiếp tục đăng ký.");
}
